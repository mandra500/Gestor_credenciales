# ============ FUNCIONES JWT ============

def generar_token(usuario_id):
    payload = {
        'usuario_id': usuario_id,
        'exp': datetime.utcnow() + app.config['JWT_ACCESS_TOKEN_EXPIRES'],
        'iat': datetime.utcnow()
    }
    token = jwt.encode(payload, app.config['JWT_SECRET_KEY'], algorithm='HS256')
    return token

def verificar_token(token):
    try:
        payload = jwt.decode(token, app.config['JWT_SECRET_KEY'], algorithms=['HS256'])
        return payload
    except jwt.ExpiredSignatureError:
        return None
    except jwt.InvalidTokenError:
        return None

def token_required(f):
    @wraps(f)
    def decorated(*args, **kwargs):
        token = None
        
        auth_header = request.headers.get('Authorization')
        if auth_header and auth_header.startswith('Bearer '):
            token = auth_header.split(' ')[1]
        else:
            token = request.cookies.get('access_token')
        
        if not token:
            flash('🔐 Por favor, inicia sesión para acceder', 'warning')
            return redirect(url_for('login'))
        
        payload = verificar_token(token)
        if not payload:
            flash('🔐 Sesión expirada. Por favor, inicia sesión nuevamente', 'warning')
            return redirect(url_for('login'))
        
        usuario = Usuario.query.get(payload['usuario_id'])
        if not usuario:
            flash('❌ Usuario no encontrado', 'error')
            return redirect(url_for('login'))
        
        return f(usuario, *args, **kwargs)
    
    return decorated

def get_backup_config():
    config = BackupConfig.query.first()
    if not config:
        config = BackupConfig()
        db.session.add(config)
        db.session.commit()
    return config

def crear_backup_full():
    try:
        config = get_backup_config()
        os.makedirs(config.carpeta_destino, exist_ok=True)
        
        fecha_str = datetime.now().strftime('%Y%m%d_%H%M%S')
        nombre_archivo = f"backup_full_{fecha_str}.db"
        ruta_archivo = os.path.join(config.carpeta_destino, nombre_archivo)
        
        db_path = app.config['SQLALCHEMY_DATABASE_URI'].replace('sqlite:///', '')
        shutil.copy2(db_path, ruta_archivo)
        
        zip_path = ruta_archivo + '.zip'
        with zipfile.ZipFile(zip_path, 'w', zipfile.ZIP_DEFLATED) as zipf:
            zipf.write(ruta_archivo, os.path.basename(ruta_archivo))
        
        os.remove(ruta_archivo)
        
        tamaño = os.path.getsize(zip_path)
        log = BackupLog(
            tipo='full',
            archivo=os.path.basename(zip_path),
            tamaño=tamaño,
            estado='exitoso'
        )
        db.session.add(log)
        
        config.ultimo_backup_full = datetime.utcnow()
        db.session.commit()
        
        limpiar_backups_antiguos()
        
        print(f"✅ Backup full creado: {zip_path}")
        return True, zip_path
    except Exception as e:
        log = BackupLog(
            tipo='full',
            estado='fallido',
            mensaje=str(e)
        )
        db.session.add(log)
        db.session.commit()
        print(f"❌ Error en backup full: {e}")
        return False, str(e)

def crear_backup_incremental():
    try:
        config = get_backup_config()
        
        if not config.ultimo_backup_full:
            crear_backup_full()
            return True, "Backup full creado como base"
        
        os.makedirs(config.carpeta_destino, exist_ok=True)
        
        fecha_str = datetime.now().strftime('%Y%m%d_%H%M%S')
        nombre_archivo = f"backup_inc_{fecha_str}.db"
        ruta_archivo = os.path.join(config.carpeta_destino, nombre_archivo)
        
        db_path = app.config['SQLALCHEMY_DATABASE_URI'].replace('sqlite:///', '')
        
        conn = sqlite3.connect(db_path)
        backup_conn = sqlite3.connect(ruta_archivo)
        conn.backup(backup_conn)
        backup_conn.close()
        conn.close()
        
        zip_path = ruta_archivo + '.zip'
        with zipfile.ZipFile(zip_path, 'w', zipfile.ZIP_DEFLATED) as zipf:
            zipf.write(ruta_archivo, os.path.basename(ruta_archivo))
        
        os.remove(ruta_archivo)
        
        tamaño = os.path.getsize(zip_path)
        log = BackupLog(
            tipo='incremental',
            archivo=os.path.basename(zip_path),
            tamaño=tamaño,
            estado='exitoso'
        )
        db.session.add(log)
        
        config.ultimo_backup_incremental = datetime.utcnow()
        db.session.commit()
        
        limpiar_backups_antiguos()
        
        print(f"✅ Backup incremental creado: {zip_path}")
        return True, zip_path
    except Exception as e:
        log = BackupLog(
            tipo='incremental',
            estado='fallido',
            mensaje=str(e)
        )
        db.session.add(log)
        db.session.commit()
        print(f"❌ Error en backup incremental: {e}")
        return False, str(e)

def limpiar_backups_antiguos():
    config = get_backup_config()
    try:
        archivos = []
        for f in os.listdir(config.carpeta_destino):
            if f.endswith('.zip') and (f.startswith('backup_full_') or f.startswith('backup_inc_')):
                ruta = os.path.join(config.carpeta_destino, f)
                archivos.append((ruta, os.path.getctime(ruta)))
        
        archivos.sort(key=lambda x: x[1], reverse=True)
        
        if len(archivos) > config.max_backups:
            for ruta, _ in archivos[config.max_backups:]:
                os.remove(ruta)
                print(f"🗑️ Backup antiguo eliminado: {ruta}")
        
        return True
    except Exception as e:
        print(f"❌ Error al limpiar backups: {e}")
        return False

def ejecutar_backup_programado():
    with app.app_context():
        config = get_backup_config()
        
        if not config.activo:
            return
        
        ahora = datetime.utcnow()
        
        if config.ultimo_backup_full:
            dias_desde_full = (ahora - config.ultimo_backup_full).days
        else:
            dias_desde_full = config.frecuencia_full + 1
        
        if dias_desde_full >= config.frecuencia_full:
            crear_backup_full()
        
        elif config.ultimo_backup_incremental:
            dias_desde_inc = (ahora - config.ultimo_backup_incremental).days
        else:
            dias_desde_inc = config.frecuencia_incremental + 1
        
        if dias_desde_inc >= config.frecuencia_incremental:
            crear_backup_incremental()
        
        db.session.commit()

def iniciar_planificador_backups():
    def planificador():
        while True:
            try:
                time.sleep(3600)
                with app.app_context():
                    ejecutar_backup_programado()
            except Exception as e:
                print(f"❌ Error en planificador de backups: {e}")
    
    hilo = threading.Thread(target=planificador, daemon=True)
    hilo.start()
    print("🔄 Planificador de backups iniciado")

def obtener_jira_client():
    """Obtiene el cliente de Jira configurado"""
    return {
        'url': app.config['JIRA_URL'],
        'auth': HTTPBasicAuth(
            app.config['JIRA_EMAIL'],
            app.config['JIRA_API_TOKEN']
        ),
        'headers': {
            'Accept': 'application/json',
            'Content-Type': 'application/json'
        }
    }

def buscar_ticket_jira(numero_ticket):
    """Busca un ticket en Jira por su número"""
    client = obtener_jira_client()
    
    try:
        response = requests.get(
            f"{client['url']}/rest/api/3/issue/{numero_ticket}",
            auth=client['auth'],
            headers=client['headers']
        )
        
        if response.status_code == 200:
            data = response.json()
            return {
                'id': data['id'],
                'key': data['key'],
                'summary': data['fields']['summary'],
                'status': data['fields']['status']['name'],
                'assignee': data['fields']['assignee']['displayName'] if data['fields']['assignee'] else None,
                'priority': data['fields']['priority']['name'] if data['fields']['priority'] else None,
                'created': data['fields']['created'],
                'updated': data['fields']['updated'],
                'description': data['fields']['description'],
                'url': f"{client['url']}/browse/{data['key']}"
            }
        return None
    except Exception as e:
        print(f"❌ Error al buscar ticket en Jira: {e}")
        return None

def actualizar_ticket_jira(ticket_key, status=None, comentario=None):
    """Actualiza un ticket en Jira"""
    client = obtener_jira_client()
    
    try:
        if status:
            response = requests.get(
                f"{client['url']}/rest/api/3/issue/{ticket_key}/transitions",
                auth=client['auth'],
                headers=client['headers']
            )
            
            if response.status_code == 200:
                transitions = response.json()
                transition_id = None
                for t in transitions['transitions']:
                    if t['name'].lower() == status.lower():
                        transition_id = t['id']
                        break
                
                if transition_id:
                    payload = {
                        'transition': {'id': transition_id}
                    }
                    requests.post(
                        f"{client['url']}/rest/api/3/issue/{ticket_key}/transitions",
                        auth=client['auth'],
                        headers=client['headers'],
                        json=payload
                    )
        
        if comentario:
            payload = {
                'body': {
                    'type': 'doc',
                    'version': 1,
                    'content': [
                        {
                            'type': 'paragraph',
                            'content': [
                                {
                                    'type': 'text',
                                    'text': comentario
                                }
                            ]
                        }
                    ]
                }
            }
            requests.post(
                f"{client['url']}/rest/api/3/issue/{ticket_key}/comment",
                auth=client['auth'],
                headers=client['headers'],
                json=payload
            )
        
        return True
    except Exception as e:
        print(f"❌ Error al actualizar ticket en Jira: {e}")
        return False

def allowed_file(filename):
    return '.' in filename and filename.rsplit('.', 1)[1].lower() in app.config['ALLOWED_EXTENSIONS']

def guardar_archivo_ppk(file):
    if file and allowed_file(file.filename):
        filename = secure_filename(file.filename)
        import time
        name, ext = os.path.splitext(filename)
        filename = f"{name}_{int(time.time())}{ext}"
        filepath = os.path.join(app.config['UPLOAD_FOLDER'], filename)
        file.save(filepath)
        return filepath
    return None

@app.route('/login', methods=['GET', 'POST'])
def login():
    token = request.cookies.get('access_token')
    if token:
        payload = verificar_token(token)
        if payload:
            usuario = Usuario.query.get(payload['usuario_id'])
            if usuario:
                flash(f'✅ ¡Bienvenido de nuevo {usuario.nombre}!', 'success')
                return redirect(url_for('dashboard'))
    
    if request.method == 'POST':
        email = request.form.get('email')
        contrasena = request.form.get('contrasena')
        
        usuario = Usuario.query.filter_by(email=email).first()
        
        if usuario and usuario.verificar_contrasena(contrasena):
            token = generar_token(usuario.id)
            
            response = redirect(url_for('dashboard'))
            response.set_cookie(
                'access_token', 
                token, 
                httponly=True, 
                secure=False,
                samesite='Lax',
                max_age=app.config['JWT_ACCESS_TOKEN_EXPIRES']
            )
            
            log_login(usuario.id)
            flash(f'✅ ¡Bienvenido {usuario.nombre}!', 'success')
            return response
        else:
            flash('❌ Email o contraseña incorrectos', 'error')
    
    return render_template('login.html')

@app.route('/logout')
def logout():
    token = request.cookies.get('access_token')
    if token:
        payload = verificar_token(token)
        if payload:
            log_logout(payload['usuario_id'])
    
    response = redirect(url_for('login'))
    response.delete_cookie('access_token')
    flash('👋 Sesión cerrada correctamente', 'info')
    return response

@app.route('/aceptar-invitacion/<token>')

@app.route('/cambiar-contrasena-invitado/<token>', methods=['GET', 'POST'])
def cambiar_contrasena_invitado(token):
    email = verificar_token_invitacion(token)
    
    if not email:
        flash('❌ El enlace ha expirado o es inválido', 'error')
        return redirect(url_for('login'))
    
    usuario = Usuario.query.filter_by(email=email).first()
    if not usuario:
        flash('❌ Usuario no encontrado', 'error')
        return redirect(url_for('login'))
    
    if not usuario.es_invitado:
        flash('ℹ️ Este usuario ya ha establecido su contraseña', 'info')
        return redirect(url_for('login'))
    
    if request.method == 'POST':
        contrasena = request.form.get('contrasena')
        confirmar = request.form.get('confirmar')
        
        if not contrasena or len(contrasena) < 6:
            flash('❌ La contraseña debe tener al menos 6 caracteres', 'error')
            return render_template('cambiar_contrasena.html', token=token, nombre=usuario.nombre)
        
        if contrasena != confirmar:
            flash('❌ Las contraseñas no coinciden', 'error')
            return render_template('cambiar_contrasena.html', token=token, nombre=usuario.nombre)
        
        usuario.contrasena_hash = bcrypt.generate_password_hash(contrasena).decode('utf-8')
        usuario.es_invitado = False
        usuario.token_invitacion = None
        usuario.token_expiracion = None
        usuario.fecha_aceptacion = datetime.utcnow()
        
        db.session.commit()
        
        flash('✅ Contraseña establecida correctamente. Ya puedes iniciar sesión.', 'success')
        return redirect(url_for('login'))
    
    return render_template('cambiar_contrasena.html', token=token, nombre=usuario.nombre)

@app.route('/dashboard')
@token_required
def dashboard(usuario):
    total_credenciales = Credencial.query.filter_by(usuario_id=usuario.id).count()
    total_aplicaciones = Aplicacion.query.count()
    total_usuarios = Usuario.query.count()
    
    ultimas_credenciales = Credencial.query.filter_by(
        usuario_id=usuario.id
    ).order_by(Credencial.fecha_creacion.desc()).limit(5).all()
    
    ultimas_actualizadas = Credencial.query.filter_by(
        usuario_id=usuario.id
    ).order_by(Credencial.fecha_actualizacion.desc()).limit(5).all()
    
    todas_credenciales = Credencial.query.filter_by(usuario_id=usuario.id).all()
    
    fechas = [c.fecha_creacion for c in todas_credenciales if c.fecha_creacion]
    df = pd.DataFrame({'fecha': fechas}) if fechas else pd.DataFrame(columns=['fecha'])
    
    if not df.empty:
        df['mes'] = df['fecha'].dt.strftime('%Y-%m')
        df['mes_nombre'] = df['fecha'].dt.strftime('%b %Y')
        
        datos_grafico = df.groupby(['mes', 'mes_nombre']).size().reset_index(name='cantidad')
        datos_grafico = datos_grafico.sort_values('mes')
        
        hoy = datetime.now()
        meses_12 = [(hoy - timedelta(days=30*i)).strftime('%Y-%m') for i in range(11, -1, -1)]
        
        meses_completos = []
        for mes in meses_12:
            registro = datos_grafico[datos_grafico['mes'] == mes]
            if not registro.empty:
                meses_completos.append({
                    'mes': mes,
                    'mes_nombre': registro.iloc[0]['mes_nombre'],
                    'cantidad': int(registro.iloc[0]['cantidad'])
                })
            else:
                fecha_obj = datetime.strptime(mes, '%Y-%m')
                meses_completos.append({
                    'mes': mes,
                    'mes_nombre': fecha_obj.strftime('%b %Y'),
                    'cantidad': 0
                })
        
        datos_grafico = meses_completos
    else:
        datos_grafico = []
    
    datos_json = json.dumps(datos_grafico)
    
    return render_template('dashboard.html',
        total_credenciales=total_credenciales,
        total_aplicaciones=total_aplicaciones,
        total_usuarios=total_usuarios,
        ultimas_credenciales=ultimas_credenciales,
        ultimas_actualizadas=ultimas_actualizadas,
        datos_grafico=datos_grafico,
        datos_json=datos_json,
        usuario=usuario
    )

@app.route('/credenciales')
@token_required
def listar_credenciales(usuario):
    """Lista todas las credenciales del usuario con filtros"""
    from sqlalchemy import or_
    
    total_credenciales = Credencial.query.filter_by(usuario_id=usuario.id).count()
    
    aplicacion_id = request.args.get('aplicacion', type=int)
    estado = request.args.get('estado', '')
    busqueda = request.args.get('buscar', '').strip()
    
    query = Credencial.query.filter_by(usuario_id=usuario.id)
    
    if aplicacion_id:
        query = query.filter_by(aplicacion_id=aplicacion_id)
    
    if estado == 'activo':
        query = query.filter(Credencial.fecha_actualizacion >= datetime.utcnow() - timedelta(days=30))
    elif estado == 'inactivo':
        query = query.filter(Credencial.fecha_actualizacion < datetime.utcnow() - timedelta(days=30))
    
    if busqueda:
        query = query.filter(
            or_(
                Credencial.nombre_servidor.contains(busqueda),
                Credencial.ip.contains(busqueda),
                Credencial.usuario.contains(busqueda),
                Credencial.descripcion.contains(busqueda)
            )
        )
    
    credenciales = query.order_by(Credencial.nombre_servidor).all()
    
    # Obtener sesiones activas para mostrar estado
    for cred in credenciales:
        cred.sesion_activa = SesionCredencial.query.filter_by(
            credencial_id=cred.id,
            estado='Activa'
        ).first()
    
    aplicaciones = Aplicacion.query.filter_by(activo=True).order_by(Aplicacion.nombre).all()
    
    return render_template('credenciales.html',
        credenciales=credenciales,
        aplicaciones=aplicaciones,
        aplicacion_seleccionada=aplicacion_id,
        estado_seleccionado=estado,
        busqueda=busqueda,
        total_credenciales=total_credenciales,
        usuario=usuario
    )

@app.route('/agregar', methods=['GET', 'POST'])
@token_required
def agregar(usuario):
    """Agregar nueva credencial"""

            aplicacion_id = request.form.get('aplicacion_id')
            
            if not aplicacion_id or aplicacion_id == '':
                aplicacion_id = identificar_aplicacion(nombre)
            
            if tipo == 'password':
                contrasena = request.form['contrasena']

                    nombre_servidor=nombre,
                    ip=ip,
                    descripcion=descripcion,
                    tipo_autenticacion=tipo,
                    usuario=usuario_cred,
                    usuario_id=usuario.id,
                    aplicacion_id=aplicacion_id if aplicacion_id else None
                )
                nueva.set_contrasena(contrasena)
                
            else:
                archivo_ppk = request.files.get('archivo_ppk')
                
                if archivo_ppk and archivo_ppk.filename != '':
                    ruta_archivo = guardar_archivo_ppk(archivo_ppk)
                    if ruta_archivo:
                        with open(ruta_archivo, 'r', encoding='utf-8') as f:
                            contenido_ppk = f.read()
                        
                        nueva = Credencial(
                            nombre_servidor=nombre,
                            ip=ip,
                            descripcion=descripcion,
                            tipo_autenticacion=tipo,
                            usuario=usuario_cred,
                            llave_publica='',
                            llave_privada=contenido_ppk,
                            ruta_archivo_ppk=ruta_archivo,
                            usuario_id=usuario.id,
                            aplicacion_id=aplicacion_id if aplicacion_id else None
                        )
                    else:
                        flash('❌ Formato de archivo no permitido. Solo .ppk, .pem, .key', 'error')
                        return render_template('agregar.html', aplicaciones=aplicaciones)
                else:
                    llave_publica = request.form.get('llave_publica', '')
                    llave_privada = request.form.get('llave_privada', '')
                    
                    if not llave_privada:
                        flash('❌ Debes subir un archivo .ppk o pegar el contenido de la llave', 'error')
                        return render_template('agregar.html', aplicaciones=aplicaciones)
                    
                    nueva = Credencial(
                        nombre_servidor=nombre,
                        ip=ip,
                        descripcion=descripcion,
                        tipo_autenticacion=tipo,
                        usuario=usuario_cred,
                        llave_publica=llave_publica,
                        llave_privada=llave_privada,
                        ruta_archivo_ppk=None,
                        usuario_id=usuario.id,
                        aplicacion_id=aplicacion_id if aplicacion_id else None
                    )
            
            db.session.add(nueva)
            db.session.commit()
            
            log_creacion(usuario.id, 'credencial', nueva.id, f'Credencial: {nueva.nombre_servidor} - IP: {nueva.ip}')
            
            app_nombre = ''
            if aplicacion_id:
                app_obj = Aplicacion.query.get(aplicacion_id)
                if app_obj:
                    app_nombre = f' ({app_obj.codigo} - {app_obj.nombre})'
            
            flash(f'✅ Credencial guardada exitosamente{app_nombre}', 'success')
            return redirect(url_for('listar_credenciales'))
            
        except Exception as e:
            db.session.rollback()
            flash(f'❌ Error al guardar: {str(e)}', 'error')
    
    return render_template('agregar.html', 
        aplicaciones=aplicaciones, 
        usuario=usuario,
        total_credenciales=total_credenciales
    )

@app.route('/aplicaciones/importar', methods=['GET', 'POST'])
@token_required
def importar_aplicaciones(usuario):
    from openpyxl import load_workbook
    
    if not usuario.es_admin:
        flash('❌ No tienes permiso para realizar esta acción', 'error')
        return redirect(url_for('dashboard'))
    
    if request.method == 'POST':
        if 'archivo_excel' not in request.files:
            flash('❌ No se seleccionó ningún archivo', 'error')
            return render_template('importar_aplicaciones.html')
        
        file = request.files['archivo_excel']
        
        if file.filename == '':
            flash('❌ No se seleccionó ningún archivo', 'error')
            return render_template('importar_aplicaciones.html')
        
        if not file.filename.endswith(('.xlsx', '.xls')):
            flash('❌ El archivo debe ser .xlsx o .xls', 'error')
            return render_template('importar_aplicaciones.html')
        
        try:
            wb = load_workbook(file)
            ws = wb.active
            
            importados = 0
            errores = 0
            errores_lista = []
            
            for row in range(2, ws.max_row + 1):
                codigo = ws.cell(row=row, column=1).value
                nombre = ws.cell(row=row, column=2).value
                descripcion = ws.cell(row=row, column=3).value
                lider_principal_nombre = ws.cell(row=row, column=4).value
                lider_principal_email = ws.cell(row=row, column=5).value
                lider_principal_telefono = ws.cell(row=row, column=6).value
                lider_alterno_nombre = ws.cell(row=row, column=7).value
                lider_alterno_email = ws.cell(row=row, column=8).value
                lider_alterno_telefono = ws.cell(row=row, column=9).value
                
                if not codigo and not nombre:
                    continue
                
                if not codigo or not nombre:
                    errores += 1
                    errores_lista.append(f"Fila {row}: Código y Nombre son obligatorios")
                    continue
                
                codigo = str(codigo).strip().upper()
                nombre = str(nombre).strip()
                
                if Aplicacion.query.filter_by(codigo=codigo).first():
                    errores += 1
                    errores_lista.append(f"Fila {row}: El código {codigo} ya existe")
                    continue
                
                nueva_app = Aplicacion(
                    codigo=codigo,
                    nombre=nombre,
                    descripcion=str(descripcion).strip() if descripcion else None,
                    activo=True
                )
                db.session.add(nueva_app)
                db.session.flush()
                
                if lider_principal_nombre or lider_principal_email or lider_principal_telefono:
                    lider_principal = LiderTecnico(
                        nombre=str(lider_principal_nombre).strip() if lider_principal_nombre else None,
                        email=str(lider_principal_email).strip() if lider_principal_email else None,
                        telefono=str(lider_principal_telefono).strip() if lider_principal_telefono else None,
                        es_alterno=False,
                        aplicacion_id=nueva_app.id
                    )
                    db.session.add(lider_principal)
                
                if lider_alterno_nombre or lider_alterno_email or lider_alterno_telefono:
                    lider_alterno = LiderTecnico(
                        nombre=str(lider_alterno_nombre).strip() if lider_alterno_nombre else None,
                        email=str(lider_alterno_email).strip() if lider_alterno_email else None,
                        telefono=str(lider_alterno_telefono).strip() if lider_alterno_telefono else None,
                        es_alterno=True,
                        aplicacion_id=nueva_app.id
                    )
                    db.session.add(lider_alterno)
                
                importados += 1
            
            db.session.commit()
            
            mensaje = f"✅ {importados} aplicaciones importadas correctamente"
            if errores > 0:
                mensaje += f" ⚠️ {errores} errores encontrados"
                flash(mensaje, 'warning')
                if errores_lista:
                    for error in errores_lista[:5]:
                        flash(f"   - {error}", 'error')
            else:
                flash(mensaje, 'success')
            
            return redirect(url_for('listar_aplicaciones'))
            
        except Exception as e:
            db.session.rollback()
            flash(f'❌ Error al importar: {str(e)}', 'error')
            return render_template('importar_aplicaciones.html')
    
    return render_template('importar_aplicaciones.html')

            fecha_hasta_dt = datetime.strptime(fecha_hasta, '%Y-%m-%d')
            query = query.filter(LogAuditoria.fecha <= fecha_hasta_dt + timedelta(days=1))
        except:
            pass
    
    logs = query.order_by(LogAuditoria.fecha.desc()).limit(200).all()
    
    acciones = db.session.query(LogAuditoria.accion).distinct().all()
    entidades = db.session.query(LogAuditoria.entidad).distinct().all()
    
    return render_template('auditoria.html',
        logs=logs,
        acciones=[a[0] for a in acciones],
        entidades=[e[0] for e in entidades],
        accion_seleccionada=accion,
        entidad_seleccionada=entidad,

@app.route('/checkin/<int:credencial_id>', methods=['GET', 'POST'])
@token_required
def checkin_credencial(usuario, credencial_id):
    """Check-in de credencial con ticket de Jira"""
    credencial = Credencial.query.get_or_404(credencial_id)
    
    # Verificar que la credencial está disponible
    sesion_activa = SesionCredencial.query.filter_by(
        credencial_id=credencial_id,
        estado='Activa'
    ).first()
    
    if sesion_activa:
        flash('❌ Esta credencial ya está en uso', 'error')
        return redirect(url_for('listar_credenciales'))
    
    if request.method == 'POST':
        ticket_key = request.form.get('ticket_key', '').strip().upper()
        
        if not ticket_key:
            flash('❌ Debes ingresar un número de ticket', 'error')
            return render_template('checkin.html', credencial=credencial)
        
        # Verificar ticket en Jira
        ticket_info = buscar_ticket_jira(ticket_key)
        
        if not ticket_info:
            flash(f'❌ Ticket {ticket_key} no encontrado en Jira', 'error')
            return render_template('checkin.html', credencial=credencial)
        
        # Verificar que el ticket está abierto
        if ticket_info['status'].lower() not in ['abierto', 'open', 'en progreso', 'in progress']:
            flash(f'❌ El ticket {ticket_key} no está activo (Estado: {ticket_info["status"]})', 'error')
            return render_template('checkin.html', credencial=credencial)
        
        # Guardar ticket en base de datos local
        ticket = Ticket.query.filter_by(numero_ticket=ticket_key).first()
        if not ticket:
            ticket = Ticket(
                numero_ticket=ticket_key,
                titulo=ticket_info['summary'],
                descripcion=ticket_info.get('description', ''),
                prioridad=ticket_info.get('priority', 'Media'),
                estado='Abierto',
                servidor=credencial.nombre_servidor,
                aplicacion_id=credencial.aplicacion_id,
                creado_por=usuario.id
            )
            db.session.add(ticket)
            db.session.commit()
        
        # Generar nueva contraseña
        nueva_contrasena = token_urlsafe(16)
        contrasena_encriptada = encriptar_contrasena(nueva_contrasena)
        
        # Crear sesión
        sesion = SesionCredencial(
            credencial_id=credencial_id,
            ticket_id=ticket.id,
            usuario_id=usuario.id,
            check_in=datetime.utcnow(),
            contrasena_usada=contrasena_encriptada,
            fecha_inicio_real=datetime.utcnow(),
            estado='Activa'
        )
        db.session.add(sesion)
        db.session.commit()
        
        # Rotar contraseña
        rotacion = RotacionContrasena(
            credencial_id=credencial_id,
            usuario_id=usuario.id,
            sesion_id=sesion.id,
            contrasena_anterior=credencial.contrasena,
            contrasena_nueva=contrasena_encriptada
        )
        db.session.add(rotacion)
        
        # Actualizar credencial
        credencial.contrasena = contrasena_encriptada
        db.session.commit()
        
        # Actualizar ticket en Jira
        actualizar_ticket_jira(
            ticket_key,
            status='En Progreso',
            comentario=f"🔐 Check-in realizado por {usuario.nombre} para la credencial {credencial.nombre_servidor}"
        )
        
        log_creacion(usuario.id, 'sesion', sesion.id, 
                     f'Check-in: Ticket {ticket_key} - Credencial {credencial.nombre_servidor}')
        
        flash(f'✅ Check-in exitoso para ticket {ticket_key}', 'success')
        flash(f'🔑 Nueva contraseña: {nueva_contrasena}', 'success')
        
        return redirect(url_for('listar_credenciales'))
    
    return render_template('checkin.html', credencial=credencial)

@app.route('/checkout/<int:sesion_id>', methods=['GET', 'POST'])
@token_required
def checkout_credencial(usuario, sesion_id):
    """Check-out de credencial"""
    sesion = SesionCredencial.query.get_or_404(sesion_id)
    credencial = sesion.credencial
    
    if sesion.usuario_id != usuario.id and not usuario.es_admin:
        flash('❌ Solo el usuario que hizo check-in puede hacer check-out', 'error')
        return redirect(url_for('listar_credenciales'))
    
    if sesion.estado != 'Activa':
        flash('❌ Esta sesión ya está cerrada', 'error')
        return redirect(url_for('listar_credenciales'))
    
    if request.method == 'POST':
        motivo = request.form.get('motivo', 'Ticket completado')
        
        sesion.check_out = datetime.utcnow()
        sesion.fecha_fin_real = datetime.utcnow()
        sesion.estado = 'Cerrada'
        sesion.motivo_cierre = motivo
        
        ticket = sesion.ticket
        if ticket:
            ticket.estado = 'Resuelto'
            ticket.fecha_cierre = datetime.utcnow()
        
        db.session.commit()
        
        if ticket:
            actualizar_ticket_jira(
                ticket.numero_ticket,
                status='Resuelto',
                comentario=f"🔐 Check-out realizado por {usuario.nombre}. Motivo: {motivo}"
            )
        
        log_creacion(usuario.id, 'sesion', sesion.id, 
                     f'Check-out: Ticket {sesion.ticket.numero_ticket} - Credencial {credencial.nombre_servidor}')
        
        flash('✅ Check-out realizado correctamente', 'success')
        return redirect(url_for('listar_credenciales'))
    
    return render_template('checkout.html', sesion=sesion, credencial=credencial)
